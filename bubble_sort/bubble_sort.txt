class Main {
    public static void bubbleSort(int[] arr){
        int n=arr.length;
        
        for(int i=0;i<n-1;i++){
            for(int j=0;j<n-i-1;j++){
                if(arr[j] > arr[j+1]){
                    int temp = arr[j] + arr[j+1];
                    arr[j]  = temp - arr[j];
                    arr[j+1] = temp - arr[j+1];
                }
            }
            
        }
        
    }
    
    public static void bubbleSortDescending(int[] arr){
        int n=arr.length;
        
        for(int i=0;i<n-1;i++){
            for(int j=0;j<n-i-1;j++){
                if(arr[j] <= arr[j+1]){
                    int temp = arr[j] + arr[j+1];
                    arr[j]  = temp - arr[j];
                    arr[j+1] = temp - arr[j+1];
                }
            }
            
        }
        
    }
    public static void main(String[] args) {
        int[] arr = {4,6,8,1,2,12};
        bubbleSort(arr);
        for(int i : arr){
            System.out.print(i + " ");
            
        }
        System.out.println();
        bubbleSortDescending(arr);
        for(int i : arr){
            System.out.print(i + " ");
            
        }
    }
}
