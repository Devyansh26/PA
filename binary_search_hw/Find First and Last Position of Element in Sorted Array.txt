class Solution {
public:
    int first(vector<int>& nums, int target) {
        int n=nums.size();
        int l=0;
        int r=n-1;
        int mid=l+((r-l)/2);
        int a=-1;
        while(l<=r){
            if(nums[mid]== target){
                a=mid;
                r=mid-1;
            }else if(nums[mid]<target){
                l=mid+1;
            }else if(nums[mid] > target){
                r=mid-1;
            }
            mid=l+((r-l)/2);
        }
        return a;
    }

    int second(vector<int>& nums, int target) {
        int n=nums.size();
        int l=0;
        int r=n-1;
        int mid=l+((r-l)/2);
        int a=-1;
        while(l<=r){
            if(nums[mid]== target){
                a=mid;
                l=mid+1;
            }else if(nums[mid]<target){
                l=mid+1;
            }else if(nums[mid] > target){
                r=mid-1;
            }
            mid=l+((r-l)/2);
        }
        return a;
    }


    vector<int> searchRange(vector<int>& nums, int target) {
        int n=nums.size();
        vector<int> result;
        result.push_back(first(nums,target));
        result.push_back(second(nums,target));

        return result;
    }   
};
