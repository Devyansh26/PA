class Main {

    public static int kSmallestElement(int[] arr,int k){
        int n=arr.length;
        if ( k<0 && k>=arr.length) return -1;
        for (int i=0;i<=k;i++) {
            int min=i;
            for(int j=i+1;j<n;j++){
                if(arr[j] < arr[min]){
                    int temp = arr[j] + arr[min];
                    arr[min] = temp - arr[min];
                    arr[j] = temp - arr[j];
                }
            }
        } 
        return arr[k];
    }
    public static void selectionSort(int[] arr) {
        int n=arr.length;
        for(int i=0;i<n-1;i++){
            int min=i;
            for(int j=i+1;j<n;j++){
                if(arr[j] < arr[min]){
                    int temp = arr[j] + arr[min];
                    arr[min] = temp - arr[min];
                    arr[j] = temp - arr[j];
                }
            }
        }   
        
    }
    public static void main(String[] args) {
        int[] arr = {4,6,8,1,2,12};
        
        selectionSort(arr);
        for(int i : arr){
            System.out.print(i + " ");
        }
        System.out.println();
        
        System.out.println(kSmallestElement(arr,3));
        
        
    }
}
